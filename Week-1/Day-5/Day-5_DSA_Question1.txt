QUEUE servers

FUNCTION addServer(server)
    servers.enqueue(server)

FUNCTION removeServer(server)
    TEMP = new Queue
    WHILE servers not empty
        s = servers.dequeue()
        IF s != server
            TEMP.enqueue(s)
    servers = TEMP

FUNCTION processRequests(requests : List<String>)
    FOR each request in requests
        server = servers.dequeue()
        PRINT request + " → " + server
        servers.enqueue(server)
servers = [S1, S2, S3]
