CLASS Flight
    VARIABLE price
    VARIABLE duration

COMPARATOR FlightComparator

    ELSE IF f1.price > f2.price RETURN 1
    ELSE IF f1.duration < f2.duration RETURN -1
    ELSE RETURN 1
